document.addEventListener("DOMContentLoaded", () => {
    // Set footer year
    const yearSpan = document.getElementById("year");
    if (yearSpan) {
        const currentYear = new Date().getFullYear();
        yearSpan.textContent = currentYear;
    }

    // Dynamic greeting on Home page
    const greetingText = document.getElementById("greeting-text");
    if (greetingText) {
        const hour = new Date().getHours();
        let greeting;

        if (hour < 12) {
            greeting = "Good morning! Welcome to my professional resume website.";
        } else if (hour < 18) {
            greeting = "Good afternoon! Thank you for taking the time to review my experience and skills.";
        } else {
            greeting = "Good evening! I appreciate your interest in my background and projects.";
        }

        greetingText.textContent = greeting;
    }

    // API Integration: Daily quote / tip
    const quoteButton = document.getElementById("load-quote");
    const quoteText = document.getElementById("quote-text");
    const quoteAuthor = document.getElementById("quote-author");

    if (quoteButton && quoteText && quoteAuthor) {
        quoteButton.addEventListener("click", async () => {
            quoteText.textContent = "Loading tip...";
            quoteAuthor.textContent = "";

            try {
                const response = await fetch("https://api.quotable.io/random");
                if (!response.ok) {
                    throw new Error("Network response was not ok");
                }
                const data = await response.json();
                quoteText.textContent = `"${data.content}"`;
                quoteAuthor.textContent = `— ${data.author}`;
            } catch (error) {
                quoteText.textContent = "Sorry, the daily tip could not be loaded at this time.";
                quoteAuthor.textContent = "";
                console.error("Error fetching quote:", error);
            }
        });
    }

    // Contact form feedback
    const contactForm = document.getElementById("contact-form");
    const formStatus = document.getElementById("form-status");

    if (contactForm && formStatus) {
        contactForm.addEventListener("submit", (event) => {
            event.preventDefault();
            formStatus.textContent =
                "Thank you for your message. This demo site has recorded your submission.";
        });
    }
});
